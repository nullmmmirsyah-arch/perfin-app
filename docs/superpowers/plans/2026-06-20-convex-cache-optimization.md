> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eliminate full table scans in getBudgetStatus, getDashboardSummary, getMonthlyTrends by caching pre-computed accumulated, unassignedCash, and monthlySpending values.

**Architecture:** New `userCaches` table stores pre-computed financial aggregates. Mutations trigger recompute via shared helper. Queries read cache + lightweight date-range queries instead of full `.collect()` scans.

**Tech Stack:** Convex (server functions, schema), TypeScript

### Task 2: Create recomputeCache helper

**Files:**
- Create: `convex/lib/recomputeCache.ts`
- Modify: `convex/_generated/api.d.ts` (auto-generated by `npx convex dev`)

This helper fetches all data for a user, recalculates accumulated, unassignedCash, monthlySpending, and upserts into userCaches. It handles both `householdId` and personal (`userId`-only) scopes.

```ts
import { MutationCtx, QueryCtx } from "../_generated/server";
import { Doc, Id } from "../_generated/dataModel";
import {
  calculateSpendingByCategory,
  calculateUnassignedCash,
  AccountMap,
  getFiscalDateDetails,
  analyzeTransactionFlow,
  parseAmount,
} from "./finance";

export async function recomputeUserCache(
  ctx: MutationCtx | QueryCtx,
  userId: string,
  householdId?: Id<"households">
): Promise<void> {
  // 1. Fetch all data
  let allTransactions: Doc<"transactions">[];
  let allAccounts: Doc<"accounts">[];
  let allCategories: Doc<"categories">[];
  let allBudgets: Doc<"budgets">[];
  let startDay = 1;

  if (householdId) {
    allTransactions = await ctx.db
      .query("transactions")
      .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
      .collect();
    allAccounts = await ctx.db
      .query("accounts")
      .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
      .collect();
    allCategories = await ctx.db
      .query("categories")
      .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
      .collect();
    allBudgets = await ctx.db
      .query("budgets")
      .withIndex("by_householdId_year_month", (q) => q.eq("householdId", householdId))
      .collect();
  } else {
    allTransactions = await ctx.db
      .query("transactions")
      .withIndex("by_userId", (q) => q.eq("userId", userId))
      .collect();
    allAccounts = await ctx.db
      .query("accounts")
      .withIndex("by_userId", (q) => q.eq("userId", userId))
      .collect();
    allCategories = await ctx.db
      .query("categories")
      .withIndex("by_userId", (q) => q.eq("userId", userId))
      .collect();
    allBudgets = await ctx.db
      .query("budgets")
      .withIndex("by_userId_year_month", (q) => q.eq("userId", userId))
      .collect();
  }

  if (householdId) {
    const household = await ctx.db.get(householdId);
    startDay = household?.budgetStartDay || 1;
  }

  const accountsMap: AccountMap = new Map(allAccounts.map((a) => [String(a._id), a]));
  const categoriesMap = new Map(allCategories.map((c) => [String(c._id), c]));

  // 2. Accumulated (all-time spending per category)
  const accumulatedByCategory = Object.entries(
    calculateSpendingByCategory(allTransactions, accountsMap, categoriesMap)
  ).map(([categoryId, amount]) => ({ categoryId, amount }));

  // 3. Monthly spending (for trends)
  const monthlyMap = new Map<string, Map<string, number>>();
  for (const tx of allTransactions) {
    const flows = analyzeTransactionFlow(tx, accountsMap, categoriesMap);
    const { year, month } = getFiscalDateDetails(tx.date, startDay);
    const key = `${year}-${String(month).padStart(2, "0")}`;
    if (!monthlyMap.has(key)) monthlyMap.set(key, new Map());
    const catMap = monthlyMap.get(key)!;
    for (const flow of flows) {
      if (flow.type === "SPENDING") {
        catMap.set(flow.categoryId, (catMap.get(flow.categoryId) || 0) + flow.amount);
      }
    }
  }
  const monthlySpending = Array.from(monthlyMap.entries())
    .map(([key, spending]) => {
      const [yearStr, monthStr] = key.split("-");
      const year = parseInt(yearStr);
      const month = parseInt(monthStr);
      const entries = Array.from(spending.entries()).map(([categoryId, amount]) => ({
        categoryId,
        amount,
      }));
      const totalSpent = entries.reduce((sum, e) => sum + e.amount, 0);
      return { year, month, spending: entries, totalSpent };
    })
    .sort((a, b) => b.year - a.year || b.month - a.month);

  // 4. Unassigned cash
  const unassignedCash = calculateUnassignedCash(
    allTransactions,
    allBudgets,
    accountsMap,
    startDay,
    categoriesMap
  );

  // 5. Upsert cache
  const existing = householdId
    ? await ctx.db
        .query("userCaches")
        .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
        .first()
    : await ctx.db
        .query("userCaches")
        .withIndex("by_userId", (q) => q.eq("userId", userId))
        .first();

  const cacheData = {
    userId,
    householdId,
    accumulatedByCategory,
    unassignedCash,
    monthlySpending,
    lastUpdatedAt: Date.now(),
  };

  if (existing) {
    await ctx.db.patch(existing._id, cacheData);
  } else {
    await ctx.db.insert("userCaches", cacheData);
  }
}

export async function getCache(
  ctx: QueryCtx | MutationCtx,
  userId: string,
  householdId?: Id<"households">
) {
  const existing = householdId
    ? await ctx.db
        .query("userCaches")
        .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
        .first()
    : await ctx.db
        .query("userCaches")
        .withIndex("by_userId", (q) => q.eq("userId", userId))
        .first();

  if (existing) return existing;

  // First-time visitor: recompute inline (one-time cost)
  await recomputeUserCache(ctx, userId, householdId);
  return householdId
    ? await ctx.db
        .query("userCaches")
        .withIndex("by_householdId", (q) => q.eq("householdId", householdId))
        .first()
    : await ctx.db
        .query("userCaches")
        .withIndex("by_userId", (q) => q.eq("userId", userId))
        .first();
}
```

First, create `convex/cronJobs.ts`:

```ts
import { internalMutation } from "./_generated/server";
import { internal } from "./_generated/api";

export const recomputeAllCaches = internalMutation({
  handler: async (ctx) => {
    // Fetch all transactions to get distinct userIds
    const txns = await ctx.db.query("transactions").collect();
    const userSet = new Set<string>();
    const householdSet = new Set<string>();
    for (const tx of txns) {
      userSet.add(tx.userId);
      if (tx.householdId) householdSet.add(tx.householdId);
    }

    let count = 0;
    for (const userId of userSet) {
      await ctx.scheduler.runAfter(0, internal.cronJobs.recomputeSingleUserCache, {
        userId,
      });
      count++;
    }

    return { users: count, households: householdSet.size };
  },
});

export const recomputeSingleUserCache = internalMutation({
  args: {
    userId: v.string(),
    householdId: v.optional(v.id("households")),
  },
  handler: async (ctx, args) => {
    const { recomputeUserCache } = await import("./lib/recomputeCache");
    await recomputeUserCache(ctx, args.userId, args.householdId);
  },
});
```

### Task 8: Rewrite `getMonthlyTrends` to use cache

**Files:**
- Modify: `convex/dashboard.ts:509-603`

Replace the full transaction scan + N+1 loop with cache read + batch category name lookup.

```ts
handler: async (ctx, { householdId, months = 3 }) => {
  const identity = await ctx.auth.getUserIdentity();
  if (!identity) throw new Error("Not authenticated");
  const userId = identity.subject;

  if (householdId && !(await ensureHouseholdAccess(ctx, householdId, userId))) {
    return [];
  }

  // Read cache
  const { getCache } = await import("./lib/recomputeCache");
  const cache = await getCache(ctx, userId, householdId);

  if (!cache || cache.monthlySpending.length === 0) {
    return [];
  }

  // Get the last N months from cache
  const now = new Date();
  const household = await getHousehold(ctx, householdId, userId);
  const startDay = household?.budgetStartDay || 1;
  const { year: currentYear, month: currentMonth } = getFiscalDateDetails(now.toISOString(), startDay);

  // Build set of months we need
  const targetMonths = new Set<string>();
  for (let i = 0; i < months; i++) {
    let targetYear = currentYear;
    let targetMonth = currentMonth - i;
    if (targetMonth < 0) { targetMonth += 12; targetYear -= 1; }
    targetMonths.add(`${targetYear}-${String(targetMonth).padStart(2, "0")}`);
  }

  // Filter cache to those months
  const filtered = cache.monthlySpending.filter(
    (m) => targetMonths.has(`${m.year}-${String(m.month).padStart(2, "0")}`)
  );

  // Resolve category names (batch query)
  const allCatIds = new Set<string>();
  for (const m of filtered) {
    for (const s of m.spending) {
      allCatIds.add(s.categoryId);
    }
  }

  let categories;
  if (householdId) {
    categories = await ctx.db.query("categories").withIndex("by_householdId", q => q.eq("householdId", householdId)).collect();
  } else {
    categories = await ctx.db.query("categories").withIndex("by_userId", q => q.eq("userId", userId)).collect();
  }
  const categoryNameMap = new Map(categories.map(c => [String(c._id), c.name]));

  // Build result
  const result = filtered.map((m) => ({
    year: m.year,
    month: m.month,
    totalSpent: m.totalSpent,
    categories: m.spending
      .map((s) => ({
        categoryId: s.categoryId,
        categoryName: categoryNameMap.get(s.categoryId) || "Unknown",
        spent: s.amount,
      }))
      .sort((a, b) => b.spent - a.spent),
  }));

  // Sort chronological
  return result.sort((a, b) => a.year - b.year || a.month - b.month);
}
```

### Task 9: Migration — Pre-compute cache for existing users

**Files:**
- Create: `convex/migrations.ts`

Create a one-time migration mutation to pre-compute caches for all existing users:

```ts
import { internalMutation } from "./_generated/server";
import { internal } from "./_generated/api";

export const seedAllCaches = internalMutation({
  handler: async (ctx) => {
    const txns = await ctx.db.query("transactions").collect();
    const userSet = new Map<string, string[]>();

    for (const tx of txns) {
      const key = tx.householdId ? `h:${tx.householdId}` : `u:${tx.userId}`;
      if (!userSet.has(key)) {
        userSet.set(key, [tx.userId, tx.householdId || ""]);
      }
    }

    let count = 0;
    for (const [, [uid, hid]] of userSet) {
      await ctx.scheduler.runAfter(0, internal.cronJobs.recomputeSingleUserCache, {
        userId: uid,
        householdId: hid ? (hid as any) : undefined,
      });
      count++;
    }

    return { cachesSeeded: count };
  },
});
```

Run from terminal:

npx convex run migrations:seedAllCaches
